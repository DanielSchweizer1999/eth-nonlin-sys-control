\section{Lyapunov Based Control Design}
\textbf{Note}

\begin{itemize}
    \item that it is preferable to know these formulas and their derivation by heart, e.g.\ on the exam.
    \item that using the general backstepping recursion~\ref{general_backstepping} is very useful and algorithmic in practice.
\end{itemize}

\subsection{Control Lyapunov Functions}\label{CLF}
Consider the stabilization problem for the following system
\begin{equation*}
    \dot{x} = f(x) + \sum_{i=1}^m g_i(x) u_i = f(x) + G(x)u,
\end{equation*}
$f:\mathbb{R}^n\to\mathbb{R}^n$, $g_i : \mathbb{R}^n \to \mathbb{R}^n$, $G : \mathbb{R}^n \to \mathbb{R}^{n \times m}$, $u : \mathbb{R} \to \mathbb{R}^m$.
\newpar{}
A function $V : \mathbb{R}^n \to \mathbb{R}$ is a control Lyapunov function (CLF) for the system above if it is differentiable, positive definite, radially unbounded, and
\begin{equation*}
    \forall x \neq 0, \quad \inf_u \underbrace{\frac{\partial V}{\partial x}(f(x) + G(x)u)}_{=:\dot{V}} < 0
\end{equation*}
For physical systems: possibility to choose $u$ such that the energy decreases. Note that we impose the infimum, as \textit{one} stabilizing best-case control input is sufficient (don't need every possible control input to be stabilizing).

\subsubsection{Conditions to Find Control Lyapunov Functions}

The condition~\ref{CLF} shown above can be hard (or impossible) to check since it involves solving an optimization problem for each $x \neq 0$. A more practical way is the following:
\newpar{}
Condition above is equivalent to
\begin{equation*}
    \forall x \neq 0: \quad \frac{\partial V}{\partial x} G(x) = 0 \Rightarrow \frac{\partial V}{\partial x} f(x) < 0
\end{equation*}
i.e.\ if we lose control at some point $x$, then the system must autonomously converge, i.e.\ $\frac{\partial V}{\partial x} f(x)<0$.
\newpar{}
The condition can also be reformulated as:
\begin{equation*}
    \forall x \neq 0: \quad \frac{\partial V}{\partial x} f(x) \geq 0 \Rightarrow \frac{\partial V}{\partial x} G(x) \neq 0,
\end{equation*}
i.e.\ for points where the system tends to diverge, we must have control in the direction of $\dot{V}$.
\subsubsection{Deriving Control Laws from CLF}
\paragraph{Artenstein Theorem}
There exists $k : \mathbb{R}^n \to \mathbb{R}^m$ (state feedback) continuous on $\mathbb{R}^n \backslash {0}$ such that the equilibrium $x^* = 0$ is globally asymptotically stable under $u = k(x)$ \textbf{iff} there exists a control Lyapunov function.
\newpar{}
\textbf{Note} that this function can have a jump at the origin which is undesireable.
\paragraph{Small Control Property}
There exists a continuous $k:\mathbb{R}^n \to \mathbb{R}^m$, smooth on $\mathbb{R}^n / {0}$, such that $x^* = 0$ is globally asymptotically stable under $u = k(x)$ \textbf{iff}
\begin{itemize}
    \item there exists a (smooth) CLF for the input-affine system~\ref{CLF}.
    \item $\forall\varepsilon>0,\exists\delta>0: \forall 0<\|x\|<\delta$, there exists
          \begin{gather*}
              u\in\mathbb{R}^m\text{ with }\|u\|<\varepsilon\\
              \text{so that } \dot{V}=L_f V(x)+L_G V(x)u<0
          \end{gather*}
\end{itemize}
This means that for any aribtrarily small input $u$ we can find an initial condition that is stabilized by that input.
\paragraph{Sontag's Formula}
Let $V: \mathbb{R}^n \to \mathbb{R}$ be a control Lyapunov function and $k$ from Sontag's formula. Then $x^*$ = 0 is \textbf{globally asymptotically stable} for $x^*=f(x)+G(x)k(x)$.
\newpar{}
To obtain the controller, fix
\begin{itemize}
    \item $c\geq 0$
    \item $a(x):=\frac{\partial V}{\partial x}f(x)\in \mathbb{R}$
    \item $b(x):=\left(\frac{\partial V}{\partial x}G(x)\right)^\top\in \mathbb{R}^{m\times 1}$
\end{itemize}
Then, Sontag's formula imposes
\begin{equation*}
    k(x):=\begin{cases}
        -cb(x)-\frac{a(x)+\sqrt{{a(x)}^2+{({b(x)}^\top b(x))}^2}}{{b(x)}^\top b(x)}b(x), & b(x)\neq0 \\
        0,                                                                               & b(x)=0
    \end{cases}
\end{equation*}

\subsection{Backstepping}
In this subsection, we show how a CLF can be constructed for the special case of systems in \textbf{strict-feedback form}, by using backstepping. It is a recursive approach that constructs a sequence of (simpler) virtual subsystems, each of which is stabilized individually.
\begin{center}
    \includegraphics[width = \linewidth]{6_integrator_bs.png}
\end{center}
\subsubsection{Vanilla Backstepping}\label{int_backstepping}
Consider the following system:
\begin{align*}
    \dot{x}_1 & = f_1(x_1) + g_1(x_1)x_2 \\
    \dot{x}_2 & = u
\end{align*}
where $x_1 \in \mathbb{R}^{n_1}$, $x_2, u \in \mathbb{R}$ (single input).
\newpar{}
We want to stabilize the origin of $\dot{x}_1$. The system can be viewed as a cascade of two subsystems: one with $x_2$ as input, and the second with $u$ as input.
\newpar{}
\textbf{Note} that the first state is the only one that is multidimensional $(n_1)$! If this form is not given, one can still try to rearrange it.

\newpar{}
\ptitle{Assumption}

We know a (smooth) state-FB control $\alpha_1 : \mathbb{R}^{n_1} \to \mathbb{R}$ and a smooth, positive definite CLF $V_1 : \mathbb{R}^{n_1} \to \mathbb{R}$ such that:
\begin{align*}
    \dot{V}_1 & = L_{f_1 + g_1 \alpha_1} V_1                                                                     \\
              & =\frac{\partial V_1}{\partial x_1} \left[f_1(x_1) + g_1(x_1)\alpha_1(x_1)\right] \leq -W_1(x_1),
\end{align*}
where $W_1(x)$ is a positive definite function, i.e.\ the origin of $x_1$ is asymptotically stable.
\newpar{}
This is the ``only time one has to think in backstepping control'' (I also had to think afterwards\dots), as fixed formulas exist for the rest of the procedure.

\paragraph{Control Approach}
\ptitle{Error Coordinates}

Using the reformulation
\begin{align*}
    \dot{x}_1 & = f_1(x_1) + g_1(x_1)\alpha_1(x_1) + g_1(x_1)[x_2 - \alpha_1(x_1)] \\
    \dot{x}_2 & = u
\end{align*}
we can introduce the \textbf{error coordinates}
\begin{equation*}
    (e_1, e_2) := (x_1 - 0, x_2 - \alpha_1(x_1))
\end{equation*}

\newpar{}
\ptitle{Error Dynamics}

\begin{align*}
    \dot{e}_1 & = f_1(e_1) + g_1(e_1)\alpha_1(e_1) + g_1(e_1)e_2            \\
    \dot{e}_2 & = u - \dot{\alpha}_1                                        \\
              & = u - \frac{\partial \alpha_1}{\partial e_1} \dot{e}_1 =: v
\end{align*}
Now, find $u = \alpha_2(x_1, x_2)$ such that $(e_1, e_2) \to 0$ as $t \to \infty$. In that case, we have for $\dot{x}_1$ the dynamics according to above reformulation which is asymptotically stable by assumption.

\newpar{}
\ptitle{Error Lyapunov Function}

Define $V_2(e_2) := \frac{1}{2}e_2^2$ and $V(e_1, e_2) := V_1(e_1) + V_2(e_2)$. Then:
{\small
\begin{gather*}
    \dot{V}(e_1, e_2) \\
    = \frac{\partial V_1}{\partial e_1}(f_1(e_1) + g_1(e_1)\alpha_1(e_1)) + \frac{\partial V_1}{\partial e_1}g_1(e_1)e_2 + e_2u - e_2\dot{\alpha}_1\\
    \leq -W_1(e_1) + \frac{\partial V_1}{\partial e_1}g_1(e_1)e_2 + e_2u - e_2\dot{\alpha}_1
\end{gather*}
}
\newpar{}
\ptitle{Control Input}\label{backstepping_input}

To make $\dot{V}$ negative definite, choose
\begin{equation*}
    u := \underbrace{-\frac{\partial V_1}{\partial e_1}g_1(e_1) + \dot{\alpha}_1}_{\text{cancelling}} \underbrace{- k_2e_2}_{\text{stabilization}} =: \alpha_2(e_1, e_2), \quad k_2 > 0
\end{equation*}
which yields
\begin{equation*}
    \dot{V}(e_1, e_2) = -W_1(e_1) - k_2 e_2^2 < 0, \quad \forall (e_1, e_2) \neq (0, 0)
\end{equation*}
i.e.\ $(e_1, e_2)$ is asymptotically stable.
\newpar{}
\ptitle{Convergence}

This result does not necessarily imply that $(x_1, x_2) = (0, 0)$ is an asymptotically stable equilibrium for the original system. In particular, in original coordinates we have
\begin{equation*}
    u = -\frac{\partial V_1}{\partial x_1}g_1(x_1) + \dot{\alpha}_1 - k_2(x_2 - \alpha_1(x_1)) = \alpha_2(x_1, x_2),
\end{equation*}
for which we see that to ensure $(x_1, x_2) = (0, 0)$ is asymptotically stable, we need
\begin{equation*}
    \alpha_1(x_1=0) = 0
\end{equation*}

\subsubsection{First Generalization}
Let us now generalize by including functions $f_2 : \mathbb{R} \times \mathbb{R} \to \mathbb{R}$ and $g_2 : \mathbb{R} \times \mathbb{R} \to \mathbb{R}$:
\begin{align*}
    \dot{x}_1 & = f_1(x_1) + g_1(x_1)x_2         \\
    \dot{x}_2 & = f_2(x_1, x_2) + g_2(x_1, x_2)u
\end{align*}
with $x_2, u_2 \in \mathbb{R}, g_2(x_1, x_2) \neq 0, \forall x_1, x_2$ and $g_2$ is known in sign and intensity.

\ptitle{Virtual Control Input}\label{virtual_control_input}

To obtain the same structure as in the vanilla backstepping case~\ref{int_backstepping}, we can use the input transformation
\begin{equation*}
    u = \frac{1}{g_2(x_1, x_2)} (v - f_2(x_1, x_2)),
\end{equation*}
we get:
\begin{align*}
    \dot{x}_1 & = f_1(x_1) + g_1(x_1)x_2 \\
    \dot{x}_2 & = v
\end{align*}
which matches the structure from~\ref{int_backstepping}.

\paragraph{Control Approach}

By following the same steps as in~\ref{int_backstepping}, we define $v$ to render $(x_1, x_2) = (0, 0)$ asymptotically stable:
\begin{align*}
    \colorbox{green!20}{$v$} & =-\frac{\partial V_1}{\partial x_1}g_1(x_1) + \dot{\alpha}_1 - k_2(x_2 - \alpha_1(x_1)) \\
    u                        & = \frac{1}{g_2(x_1, x_2)}\left(\colorbox{green!20}{$v$} - f_2(x_1, x_2)\right)
\end{align*}
Note that $u=\alpha_2$ for~\ref{three_state_bs}.

\subsubsection{Generalization to Three States}\label{three_state_bs}

\ptitle{Preliminary Note}

It may be easier to directly use the general backstepping recursion, but here we show the three-state case for completeness.

\newpar{}
\ptitle{System}

In the following we see backstepping through two integrators:
\begin{align*}
    \dot{x}_1 & = f_1(x_1) + g_1(x_1)x_2                   \\
    \dot{x}_2 & = f_2(x_1, x_2) + g_2(x_1, x_2)x_3         \\
    \dot{x}_3 & = f_3(x_1, x_2, x_3) + g_3(x_1, x_2, x_3)u
\end{align*}
Assume: $g_2, g_3 \ne 0, \, \forall x_1, x_2, x_3$, and both their value and their sign are known.
\paragraph{Control Approach}
\ptitle{Error Coordinates}

In the part above, we showed that there exists $u=\alpha_2$ such that for $x_3 = \alpha_2(x_1, x_2)$, $(e_1, e_2) \to 0$. This means that $\alpha_2$ is the goal of our current iteration.
\newpar{}
In other words, we want to define the new input $u$ so that $x_3$ approaches $\alpha_2$, and simultaneously $x_3 \to 0$. Thus, we introduce the error coordinate $e_3 := x_3 - \alpha_2(x_1, x_2)$.
\newpar{}
\ptitle{Virtual Control Input}

As in~\ref{virtual_control_input}, we can use
\begin{equation*}
    u = \frac{1}{g_3(x_1, x_2, x_3)}\left(v - f_3(x_1, x_2, x_3)\right),
\end{equation*}
to obtain
\begin{align*}
    \dot{x}_1 & = f_1(x_1) + g_1(x_1)x_2           \\
    \dot{x}_2 & = f_2(x_1, x_2) + g_2(x_1, x_2)x_3 \\
    \dot{x}_3 & = v,
\end{align*}
and therefore we can apply integrator backstepping once more.

\newpar{}
\ptitle{Error Dynamics}

In error coordinates, the system reads:
\begin{align*}
    \dot{e}_1 & = f_1(e_1) + g_1(e_1)(e_2 + \alpha_1(e_1))                                                 \\
    \dot{e}_2 & = f_2(e_1, e_2 + \alpha_1(e_1))                                                            \\
              & + g_2(e_1, e_2 + \alpha_1(e_1))(e_3 + \alpha_2(e_1, e_2 + \alpha_1(e_1))) - \dot{\alpha}_1 \\
    \dot{e}_3 & = v - \dot{\alpha}_2
\end{align*}
\newpar{}
\ptitle{Lyapunov Function}

Now, define
\begin{align*}
    V_3(e_3)         & := \frac{1}{2}e_3^2               \\
    V(e_1, e_2, e_3) & := V_1(e_1) + V_2(e_2) + V_3(e_3)
\end{align*}

One can show that
\begin{equation*}
    \dot{V} = L_{f_1 + g_1\alpha_1}V_1(e_1) - k_2e_2^2 + e_2g_2(e_1, e_2 + \alpha_1(e_1))e_3 + e_3(v - \dot{\alpha}_2)
\end{equation*}

Since we want to ensure that $\dot{V}$ is negative definite, then we can choose the variable $v$ based on the same reasoning discussed above, i.e., with some "cancelling terms", and some "stabilizing terms":
\begin{equation*}
    v = -\underbrace{e_2g_2(e_1, e_2 + \alpha_1(e_1)) + \dot{\alpha}_2}_{\text{``cancelling terms''}} - \underbrace{k_3e_3}_{\text{``stabilizing term''}}
\end{equation*}
\newpar{}
\ptitle{Control Input}

Then, in original coordinates:
{\tiny
\begin{equation*}
    u  = \frac{- \overbrace{(x_2 - \alpha_1(x_1))}^{e_2}g_2(x_1, x_2) + \dot{\alpha}_2 - k_3(x_3 - \alpha_2(x_1, x_2)) - f_3(x_1, x_2, x_3)}{g_3(x_1, x_2, x_3)}
\end{equation*}
}
\begin{equation*}
    =: \alpha_3(x_1, x_2, x_3)
\end{equation*}
Note that $\alpha_2$ is contained in the formula, which intuitively means that we also track the inner references using this $u$.
\subsubsection{General Backstepping Recursion}\label{general_backstepping}

Based on the two generalizations shown above, it is now possible to employ the backstepping technique to a general system in \textbf{strict-feedback form}:
\begin{align*}
    \dot{x}_1 & = f_1(x_1) + g_1(x_1)x_2                         \\
    \dot{x}_2 & = f_2(x_1, x_2) + g_2(x_1, x_2)x_3               \\
              & \vdots                                           \\
    \dot{x}_k & = f_k(x_1, \ldots, x_k) + g_k(x_1, \ldots, x_k)u
\end{align*}

$x_1 \in \mathbb{R}^{n_1}, \, x_2, \ldots, x_k, u \in \mathbb{R}, \, g_2, \ldots, g_k \text{ nowhere zero}.$

The functions $f_i, g_i$ are (sufficiently) smooth, because their derivatives are needed in the definitions of $\alpha_i$.
\newpar{}
\textbf{Note}: The recursion makes it obvious that backstepping is very \textbf{sensitive} to errors in $f_i, g_i$!

\newpar{}
\ptitle{Backstepping Recursion}

\begin{enumerate}
    \item Input data: A CLF $V_1$ for the first system $\dot{x}_1 = f_1(x_1) + g_1(x_1)u_1$, with a smooth feedback $\alpha_1$ that asymptotically stabilizes the origin for $u_1 = \alpha_1(x_1)$.
    \item For $i = 2, \ldots, k$:
          Construct a CLF $V_i(e_i) = \frac{1}{2}e_i^2$, define
          \begin{equation*}
              V = \sum_{j=1}^i V_j,
          \end{equation*}
          and a feedback $\alpha_i$ that asymptotically stabilizes
              {\small
                  \begin{equation*}
                      (e_1, \ldots, e_i) = (x_1-0, x_2-\alpha_1(x_1), \ldots, x_i-\alpha_{i-1}(x_1,\ldots,x_{i-1}))
                  \end{equation*}
              }
          with (\textbf{most important formula} in backstepping!)
          {\footnotesize
                  \begin{gather*}
                      \alpha_i(x_1, \ldots, x_i) \\
                      = \frac{1}{g_i}\left( -\frac{\partial V_{i-1}}{\partial e_{i-1}}g_{i-1} + \dot{\alpha}_{i-1} - k_i(x_i - \alpha_{i-1}(x_1, \ldots, x_{i-1})) - f_i \right)
                  \end{gather*}
              }
    \item Finally, the input is chosen as:
          \begin{equation*}
              u := \alpha_k(x_1, \ldots, x_k)
          \end{equation*}
\end{enumerate}
\newpar{}
\ptitle{Remarks}

The function
\begin{equation*}
    V = \sum_{i=1}^k V_i
\end{equation*}
with $V_i(e_i) = \frac{1}{2}e_i^2$ for $i = 2, \ldots, k$, and distinct $V_1$, is a Control Lyapunov Function.

\paragraph{Disturbance Rejection}
In the part above, we only focused on the case where the system is not affected by disturbances. However, it is important to remark that backstepping can also be used with systems affected by \textbf{bounded disturbances}.
