\subsection{Terminology}
\ptitle{Compactness}

Compact sets are \textit{closed and bounded}. Examples:
\begin{itemize}
    \item $(-\infty, 0]$ not compact as neither closed nor bounded          % ChkTeX 9
    \item $(-1,0]$ not compact as not closed                                % ChkTeX 9
    \item $[-1,0]$ compact as closed and bounded
\end{itemize}

\newpar{}
\ptitle{Piecewise Continuity}

At most jumps at discrete points, continuous in between.

\newpar{}
\ptitle{Boundedness}

A set is bounded if all of its points lie within a certain distance of each other.

\newpar{}
\ptitle{Radial Unboundedness}

A function is radially unbounded, if $\forall c\in \mathbb{R} > 0, \exists r>0 | \; \forall x\in \mathbb{R}^n :[\lVert x \rVert > r \implies f(x) > c ]$

\newpar{}
\ptitle{Smoothness}

A function is smooth if it is infinitely differentiable and the derivatives are continuous.

\subsection{Linear Algebra}
\textbf{Null Space}: $A\cdot x = 0$ is the set of vectors $x$ that is mapped onto the origin. Solve the HLGS.

\textbf{Image}: $A \cdot x = y$ is a vector space spanned by all vectors $y$ that can be reached through the matrix $A$. Find linear independent columns of $A$ (Gauss on $A^T$).

\textbf{Def}: A matrix $A$ is called \textbf{scew-symmetric} iff $A = -A^T$. One property of these matrices is that $x^T A x = 0, \; \forall x\in \mathbb{R}^n$

\subsection{Ruth-Hurwitz Criterion}
A \textbf{square} matrix $A$ is called \textbf{Hurwitz} iff $Re(\lambda_i) < 0 \; \forall \lambda \in SPAN\{ A \}$ (eigenvalues have strictly negative real parts)

Given the following polynomial of degree $n$:
\begin{align*}
    D(s) & = a_n s^n + a_{n-1}s^{n-1} + \cdots + a_1s^1 + a_0s^0 \\
         & = a_n s^n + a_{n-1}s^{n-1} + \cdots + a_1s + a_0
\end{align*}
All poles are strictly in the left half plane (negative real part) iff the following table only contains positive elements. Example for a polynomial of degree 4:
\begin{align*}
    \begin{array}{|c|c|c|c|}
        \hline
        a_0                                                         & a_2 & a_4 & 0 \\
        \hline
        a_1                                                         & a_3 & 0   & 0 \\
        \hline
        \frac{a_1 a_2 - a_0 a_3}{a_1}                               & a_4 & 0   & 0 \\
        \hline
        \frac{(a_1 a_2 - a_0 a_3)a_3 - a_1^2a_4}{a_1 a_2 - a_0 a_3} & 0   & 0   & 0 \\
        \hline
        a_4                                                         & 0   & 0   & 0 \\
        \hline
    \end{array}
\end{align*}

\subsection{Characterizations of Passivity for Linear Systems}\label{passivity_linear_sys}

Passive systems are especially significant in the context of linear systems because their passivity can be characterized \textbf{both} in the frequency and time domains.

\subsubsection{Positive Real Transfer Functions}

A $p \times p$ proper transfer function matrix $G(s)$ is called positive real if all the following conditions hold:

\begin{enumerate}
    \item The poles of all elements of $G(s)$ have non-positive real parts.
    \item For all real frequencies $\omega$ where $j\omega$ is not a pole of any element of $G(s)$, the matrix:

          \begin{equation*}
              G(j\omega) + G(-j\omega)^T
          \end{equation*}

          is positive semi-definite.

    \item Any pure imaginary pole $j\omega$ of any element of $G(s)$ is a simple pole, and the residue matrix:

          \begin{equation*}
              \lim_{s \to j\omega} (s - j\omega) G(s)
          \end{equation*}

          is positive semi-definite Hermitian.
\end{enumerate}
\newpar{}
$G(s)$ is called \textbf{strictly positive real} if $G(s - \epsilon)$ is positive real for some $\epsilon > 0$.

\newpar{}
\ptitle{1D Case}

For $p = 1$, the second condition reduces to:

\begin{equation*}
    \operatorname{Re}(G(j\omega)) \geq 0, \quad \forall \omega \in \mathbb{R},
\end{equation*}

which holds only if the relative degree of $G(s)$ is at most one.

\paragraph{Strictly Positive Real Transfer Functions}

Let $G(s)$ be a proper rational $p \times p$ transfer function matrix. Suppose that $\det(G(s) + G(-s)^T)$ is not identically zero for all $s$. Then, $G(s)$ is strictly positive real \textbf{if and only if} the following three conditions hold:

\begin{enumerate}
    \item $G(s)$ is Hurwitz,
    \item $G(j\omega) + G(-j\omega)^T$ is positive definite $\forall \omega \in \mathbb{R}$,
    \item Either $G(\infty) + G(\infty)^T$ is positive definite, or it is positive semi-definite and:

          \begin{equation*}
              \lim_{\omega \to \infty} \omega^2 M^T (G(j\omega) + G(-j\omega)^T) M
          \end{equation*}

          is positive definite for any full-rank $p \times (p - q)$ matrix $M$ such that:

          \begin{equation*}
              M^T (G(\infty) + G(\infty)^T) M = 0,
          \end{equation*}

          where $q = \operatorname{rank}(G(\infty) + G(\infty)^T)$.
\end{enumerate}
\newpar{}
If $\epsilon = 0$ then the transfer function $G(s)$ is \textbf{simply positive real}.

\newpar{}
\ptitle{Kalman-Yakubovich-Popov (KYP) Lemma}

Consider the $m \times m$ transfer function matrix:

\begin{equation*}
    G(s) = C(sI - A)^{-1}B + D,
\end{equation*}

where the pair $(A, B)$ is controllable and the pair $(A, C)$ is observable.
\newpar{}
$G(s)$ is strictly positive real \textbf{if and only if} there exist matrices $P = P^T > 0$, $L$, $W$, and $\epsilon > 0$ such that:

\begin{align*}
    PA + A^T P & = -L^T L - \epsilon P, \\
    PB         & = C^T - L^T W,         \\
    W^T W      & = D + D^T.
\end{align*}

\paragraph{LTI Passivity}

The LTI system:

\begin{equation*}
    \dot{x} = Ax + Bu, \quad y = Cx + Du,
\end{equation*}

with transfer matrix $G(s) = C(sI - A)^{-1}B + D$ is:

\begin{itemize}
    \item passive if $G(s)$ is positive real,
    \item strictly passive if $G(s)$ is strictly positive real.
\end{itemize}
